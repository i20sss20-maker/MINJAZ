function offerRank(items,p){
 const prices=items.map(x=>Number(x.price||0)).filter(Boolean),hours=items.map(x=>Number(x.delivery_hours||0)).filter(Boolean);const minP=Math.min(...prices,Number(p.price||0)),maxP=Math.max(...prices,Number(p.price||0)),minH=Math.min(...hours,Number(p.delivery_hours||0)),maxH=Math.max(...hours,Number(p.delivery_hours||0));
 const priceScore=maxP===minP?100:100-((Number(p.price||0)-minP)/(maxP-minP))*100,fastScore=maxH===minH?100:100-((Number(p.delivery_hours||0)-minH)/(maxH-minH))*100,ratingScore=Math.min(100,(Number(p.rating||0)/5)*100),history=Math.min(100,Number(p.completed_tasks||0)*4),ontime=Math.min(100,Number(p.on_time_rate||0));
 return Math.round(priceScore*.25+fastScore*.2+ratingScore*.25+ontime*.2+history*.1)
}
