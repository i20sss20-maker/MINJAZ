def transition_order_revision(oid,client_id,note):
    c=db_connect()
    try:
        c.autocommit=False
        with c.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
            cur.execute("""select o.*,t.status task_status,coalesce(p.revisions,0)::int revisions_allowed
                           from orders o join tasks t on t.id=o.task_id left join proposals p on p.id=o.proposal_id
                           where o.id=%s and o.client_id=%s for update of o,t""",(int(oid),int(client_id)))
            o=cur.fetchone()
            if not o:
                c.rollback();return None,None,None,'order_not_found'
            if o.get('status')!='delivered':
                c.rollback();return None,dict(o),None,'invalid_order_state'
            cur.execute("select id from order_cancellation_requests where order_id=%s and status in ('pending','in_review') limit 1",(int(oid),))
            if cur.fetchone():
                c.rollback();return None,dict(o),None,'active_cancellation_exists'
            cur.execute('select count(*)::int used from order_revision_requests where order_id=%s',(int(oid),))
            used=int((cur.fetchone() or {}).get('used') or 0); allowed=int(o.get('revisions_allowed') or 0)
            if used>=allowed:
                c.rollback();return None,dict(o),{'allowed':allowed,'used':used},'revision_limit_reached'
            sequence=used+1
            cur.execute("insert into order_revision_requests(order_id,requested_by,sequence_no,note) values(%s,%s,%s,%s) returning *",(int(oid),int(client_id),sequence,str(note)[:3000]))
            rr=dict(cur.fetchone())
            cur.execute("insert into messages(order_id,sender_id,body) values(%s,%s,%s)",(int(oid),int(client_id),('طلب تعديل '+str(sequence)+'/'+str(allowed)+': '+str(note))[:4000]))
            cur.execute("update orders set status='revision_requested' where id=%s",(int(oid),))
            cur.execute("update tasks set status='in_progress',updated_at=now() where id=%s",(o['task_id'],))
            c.commit()
            return {k:as_json(v) for k,v in rr.items()},dict(o),{'allowed':allowed,'used':sequence,'remaining':max(allowed-sequence,0)},None
    except Exception:
        c.rollback();raise
    finally:
        c.close()

def session_token_from_headers(headers):
    h=str(headers.get('Authorization') or '').strip()
    if h.startswith('Bearer '):
        raw=h[7:].strip()
        if raw:return raw,'bearer'
    cookie=str(headers.get('Cookie') or '')
    m=re.search(r'(?:^|;\\s*)'+re.escape(SESSION_COOKIE_NAME)+r'=([A-Fa-f0-9]{64})(?:;|$)',cookie)
    return (m.group(1),'cookie') if m else ('','')
